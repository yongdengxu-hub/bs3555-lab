<script>
function n(x, d) { return (x === '' || x === null || x === undefined || isNaN(x)) ? '—'
  : Number(x).toLocaleString('en-GB', {minimumFractionDigits: d || 0, maximumFractionDigits: d || 0}); }

var KEY = 'bs3555player';
var $ = function (id) { return document.getElementById(id); };

function load(code) {
  $('gate').style.display = 'none';
  $('loading').style.display = '';
  google.script.run.withSuccessHandler(render).withFailureHandler(function (e) {
    $('loading').innerHTML = '<div class="err">Could not load: ' + e.message + '</div>';
  }).getMyState(code);
}

/** Back to the code box, optionally with a reason. */
function gate(msg) {
  $('loading').style.display = 'none';
  $('app').style.display = 'none';
  $('chg').style.display = 'none';
  $('gate').style.display = '';
  if (msg) { $('gateErr').textContent = msg; $('gateErr').style.display = ''; }
  else { $('gateErr').style.display = 'none'; }
  $('code').focus();
}

$('go').onclick = function () {
  var v = $('code').value.trim().toUpperCase();
  if (!v) { gate('Type your code first.'); return; }
  localStorage.setItem(KEY, v);
  load(v);
};
$('code').addEventListener('keydown', function (ev) {
  if (ev.key === 'Enter') $('go').click();
});
$('chg').onclick = function (ev) {
  ev.preventDefault();
  localStorage.removeItem(KEY);
  $('code').value = '';
  gate('');
  return false;
};

var saved = localStorage.getItem(KEY);
if (saved) load(saved); else gate('');

function render(s) {
  // An unrecognised code is a typo, not a failure — send them back to the box.
  if (!s.found) { gate(s.error || 'Code not recognised.'); return; }

  $('loading').style.display = 'none';
  $('app').style.display = '';
  $('chg').style.display = '';

  document.getElementById('msg').innerHTML = s.error ? '<div class="err">' + s.error + '</div>' : '';
  document.getElementById('asof').textContent =
    'Prices as of ' + s.latestDate + (s.spot ? '  ·  USD ' + s.spot.usd.toFixed(4) +
    ' · EUR ' + s.spot.eur.toFixed(4) + ' GBP per unit  ·  USD realised vol ' +
    (100 * s.spot.vu).toFixed(1) + '%' : '');

  if (!s.found) { renderLb(s); return; }

  var m = s.mandate;
  document.getElementById('mandate').innerHTML = !m
    ? '<div class="mand">No round is open yet. Your first mandate is issued on the start date below.</div>'
    : (m.open
      ? '<div class="mand"><b>Round ' + m.round + '.</b> You ' +
        (m.isPayable
          ? 'must pay <b>' + n(m.qty) + ' ' + m.ccy + '</b>'
          : 'will receive <b>' + n(m.qty) + ' ' + m.ccy + '</b>') +
        ' in <b>' + m.daysLeft + ' trading days</b>. Fixing it with a forward today is worth <b>£' +
        n(m.lockNow, 2) + '</b>. On your books it currently marks at <b>£' + n(m.value, 2) +
        '</b>. Hedge it by <b>' + (m.isPayable ? 'buying' : 'selling') + ' ' + m.ccy +
        ' forward</b>, or with a ' + (m.isPayable ? 'call' : 'put') +
        '. Your job is to make the outcome <i>certain</i>.</div>'
      : '<div class="mand"><b>Round ' + m.round + ' mandate settled.</b> ' +
        (m.cost >= 0 ? 'It cost you <b>£' + n(m.cost, 2) + '</b>.'
                     : 'It brought in <b>£' + n(-m.cost, 2) + '</b>.') +
        ' Your score is the certainty you achieved, not the profit.</div>');

  document.getElementById('sched').innerHTML = (s.schedule && s.schedule.length)
    ? '<tr><th>round</th><th>from</th><th>to</th><th>mandate</th></tr>' +
      s.schedule.map(function (r) {
        var live = m && r.round === m.round;
        return '<tr class="' + (live ? 'me' : '') + '"><td>' + r.round + '</td><td>' + r.start +
        '</td><td>' + r.end + '</td><td>' + r.mandate + (live ? '  ← now' : '') + '</td></tr>';
      }).join('')
    : '';

  var rr = s.rounds || [];
  document.getElementById('rounds').innerHTML = rr.length
    ? '<tr><th>#</th><th>mandate</th><th>net worth</th><th>vol</th><th>max DD</th><th>hedge eff.</th><th>trades</th></tr>' +
      rr.map(function (r) {
        return '<tr><td>' + r.round + '</td><td>' + r.mandate + '</td><td>£' + n(r.netWorth, 2) +
        '</td><td>' + r.vol + '%</td><td class="bad">' + r.maxDD + '%</td><td class="' +
        (r.hedgeEff >= 80 ? 'ok' : '') + '">' + (r.hedgeEff === '' ? '—' : r.hedgeEff + '%') +
        '</td><td>' + r.trades + '</td></tr>';
      }).join('')
    : '<tr><td class="muted">No completed rounds yet.</td></tr>';

  var a = s.account;
  document.getElementById('kpis').innerHTML =
    kpi('net worth', '£' + n(a.nav, 2), a.nav >= 0 ? '' : 'bad') +
    kpi('GBP cash', '£' + n(a.cash, 2)) +
    kpi('USD balance', n(a.usd, 2)) +
    kpi('net FX exposure', '£' + n(a.exposure, 0)) +
    kpi('margin used', '£' + n(a.margin, 0)) +
    kpi('leaderboard rank', s.myRank || '—');

  var p = s.positions;
  document.getElementById('pos').innerHTML = p.length
    ? '<tr><th>id</th><th>instrument</th><th>size</th><th>entry</th><th>strike</th><th>days left</th><th>value</th></tr>' +
      p.map(function (r) {
        return '<tr><td>' + r.posId + '</td><td>' + r.kind + ' ' + r.ccy + '</td><td>' + n(r.qty) +
        '</td><td>' + r.entry + '</td><td>' + (r.strike || '—') + '</td><td>' + r.daysLeft +
        '</td><td class="' + (r.value >= 0 ? 'ok' : 'bad') + '">£' + n(r.value, 2) + '</td></tr>';
      }).join('')
    : '<tr><td class="muted">No open positions. Your payable is unhedged.</td></tr>';

  document.getElementById('ord').innerHTML =
    '<tr><th>submitted</th><th>order</th><th>size</th><th>status</th><th>filled</th><th>note</th></tr>' +
    (s.orders.length ? s.orders.map(function (o) {
      return '<tr><td>' + o.when + '</td><td>' + o.dir + ' ' + o.kind + ' ' + o.ccy +
      '</td><td>' + n(o.qty) + '</td><td class="' + (o.status === 'FILLED' ? 'ok' :
      (o.status === 'REJECTED' ? 'bad' : '')) + '">' + o.status + '</td><td>' +
      (o.fillDate || '—') + (o.fillPx ? ' @' + o.fillPx : '') + '</td><td class="muted">' +
      (o.note || '') + '</td></tr>';
    }).join('') : '<tr><td class="muted">No orders yet.</td></tr>');

  renderLb(s);
  spark(s.nav);
}

function kpi(label, val, cls) {
  return '<div class="kpi"><small>' + label + '</small><b class="' + (cls || '') + '">' + val + '</b></div>';
}

function renderLb(s) {
  document.getElementById('lb').innerHTML =
    '<tr><th>#</th><th>student</th><th>rounds</th><th>avg hedge eff.</th><th>worst round</th>' +
    '<th>avg vol</th><th>per round</th><th></th></tr>' +
    (s.leaderboard.length ? s.leaderboard.map(function (r) {
      return '<tr class="' + (r.isMe ? 'me' : '') + '"><td>' + r.rank + '</td><td>' + r.alias +
      (r.isMe ? ' (you)' : '') + '</td><td>' + r.roundsDone + '</td><td>' +
      (r.avgHedgeEff === '' ? '—' : r.avgHedgeEff + '%') + '</td><td>' +
      (r.worstRoundEff === '' ? '—' : r.worstRoundEff + '%') + '</td><td>' + r.avgVol +
      '%</td><td class="muted">' + (r.perRound || '') + '</td><td class="muted">' +
      r.status + '</td></tr>';
    }).join('') : '<tr><td class="muted">Leaderboard is empty until the first round closes.</td></tr>');
}

function spark(nav) {
  var cv = document.getElementById('spark');
  if (!nav || nav.length < 2) { cv.style.display = 'none'; return; }
  var ctx = cv.getContext('2d'), W = cv.width = cv.offsetWidth, H = cv.height = 120;
  var vs = nav.map(function (p) { return p.v; });
  var lo = Math.min.apply(null, vs), hi = Math.max.apply(null, vs), pad = (hi - lo) * 0.15 || 1;
  lo -= pad; hi += pad;
  ctx.clearRect(0, 0, W, H);
  ctx.strokeStyle = '#8A1538'; ctx.lineWidth = 2; ctx.beginPath();
  vs.forEach(function (v, k) {
    var x = k / (vs.length - 1) * (W - 4) + 2, y = H - 4 - (v - lo) / (hi - lo) * (H - 12);
    k ? ctx.lineTo(x, y) : ctx.moveTo(x, y);
  });
  ctx.stroke();
  ctx.fillStyle = '#5E6A75'; ctx.font = '11px system-ui';
  ctx.fillText('net worth £' + vs[vs.length - 1].toFixed(0), 4, 12);
}
